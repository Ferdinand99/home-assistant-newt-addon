# Home Assistant Add-on AppArmor profile for Newt
#
# This profile is enabled by default in the beta channel (apparmor: true in config.yaml).
# The stable channel keeps apparmor disabled for hosts without AppArmor support.
#
# The profile name below does not have to match the add-on slug: Supervisor rewrites
# it to the slug via adjust_profile() before loading, so leave it as-is.
#
# Scope note: this profile targets Newt's default userspace (netstack) WireGuard mode.
# The native-interface modes (USE_NATIVE_INTERFACE / USE_NATIVE_MAIN_INTERFACE, and
# client mode) additionally need netlink and sysctl writes under /proc/sys/net, which
# are deliberately denied below. Use the stable channel if you need those modes.

#include <tunables/global>

profile hassio-addons-newt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Required runtime capabilities for Newt networking behavior.
  capability net_admin,
  capability net_raw,

  # General network access for outbound tunnel connectivity.
  network,

  # Runtime binaries/scripts.
  /usr/bin/env ix,
  /bin/bash ix,
  /usr/bin/bash ix,
  /usr/bin/jq ix,
  /bin/rm ix,
  /bin/busybox ix,
  /run.sh rix,
  /usr/bin/newt ixr,

  # Read add-on options and allow Newt to persist its own config under /data.
  /data/** rwk,
  /etc/ssl/** r,
  /etc/hosts r,
  /etc/resolv.conf r,

  # Health file and transient runtime files.
  /tmp/** rw,

  # TUN device, needed if a native WireGuard interface is used instead of netstack.
  /dev/net/tun rw,

  # Deny dangerous kernel/module operations.
  deny capability sys_admin,
  deny capability sys_module,
  deny capability sys_ptrace,

  # Deny mount and sensitive proc/sys writes.
  deny mount,
  deny /proc/** w,
  deny /sys/** w,
}
